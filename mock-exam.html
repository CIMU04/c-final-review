三、看代码写结果

1. 20, 50　p 指向 a[2]：*(p-1) = a[1] = 20；p[2] = *(p+2) = a[4] = 50。下标是相对 p 的偏移，不是相对数组首元素。

2. 28　s 只在第一次调用时初始化为 2，之后保留：3、5、8、12，累加 3+5+8+12 = 28。

3. 5, 27　宏只做文本替换：SQ(n+1) → n + 1 * n + 1 = 2+2+1 = 5；CUBE 参数与整体都加了括号 → 3*3*3 = 27。所以宏参数、整体都要加括号，且不要传带 ++ 的实参。

4. 78, 12　小端把最低有效字节放在低地址：内存依次为 78 56 34 12，故 b[0]=0x78、b[3]=0x12。

四、代码找错题

第 1 题（6 分）　三处问题：① dest 是形参副本，dest = src 只改了局部变量，一个字符都没拷贝，主调函数完全不受影响（这是根本错误，不是「少写了 '\0'」）；② 函数体内没有任何逐字符复制，也没有复制结尾 '\0'；③ str1 未初始化就用 %s 输出，行为未定义。另可指出 src 应加 const、dest 空间需足够。
改正： 

void my_strcpy(char *dest, const char *src) {
    while ((*dest++ = *src++) != '\0')
        ;                 /* 赋值表达式已把 '\0' 一起复制 */
}

第 2 题（6 分）　两处问题：① s[0].score = {90, 85, 88}; 不合法——数组名不可被整体赋值，初始化列表只能在定义变量时使用；应逐元素赋值或用循环 / memcpy。② 用了 strcpy 和 printf 却缺少 #include <string.h> 与 #include <stdio.h>。
改正： 

#include <stdio.h>
#include <string.h>
/* struct Stu 定义不变 */
int main(void) {
    struct Stu s[3] = {0};            /* 顺手初始化，避免垃圾值 */
    strcpy(s[0].name, "张三");
    s[0].id = 1;
    { int j; float v[3] = {90, 85, 88};
      for (j = 0; j < 3; j++) s[0].score[j] = v[j]; }
    printf("%s %d %.1f\n", s[0].name, s[0].id, s[0].score[0]);
    return 0;
}

五、编程题 · 参考实现

#include <stdio.h>

struct Student {
    char  name[20];
    int   id;
    float scores[3];
};

/* 8 分：p 指向数组首元素，用 p[i] 或 (p+i)-> 访问第 i 个学生 */
void inputStudent(struct Student *p, int n)
{
    int i, j;
    for (i = 0; i < n; i++) {
        printf("请输入第 %d 个学生的姓名: ", i + 1);
        scanf("%19s", p[i].name);
        printf("请输入学号: ");
        scanf("%d", &p[i].id);
        for (j = 0; j < 3; j++) {
            printf("请输入第 %d 门课成绩: ", j + 1);
            scanf("%f", &p[i].scores[j]);
        }
    }
}

/* 10 分：先用下标记录最大值位置，全程不改动原数组 */
void printMaxScore(struct Student stuArr[], int n)
{
    int i, j, max = 0;
    float sum, maxSum = -1.0f;
    for (i = 0; i < n; i++) {                /* 不写 i+1，避免越界 */
        sum = 0;
        for (j = 0; j < 3; j++) sum += stuArr[i].scores[j];
        if (sum > maxSum) { maxSum = sum; max = i; }
    }
    printf("总分最高者: 姓名 %s  学号 %d  总分 %.1f\n",
           stuArr[max].name, stuArr[max].id, maxSum);
}

int main(void)                                /* 6 分 */
{
    struct Student stu[2];
    inputStudent(stu, 2);
    printMaxScore(stu, 2);
    return 0;
}

评分要点：结构体定义 4 分（不要写 typedef struct Student {...}; 这种没有别名的写法）；输入函数 8 分（成员访问、& 取地址、3 门课循环）；最高分函数 10 分（求 3 门总分、打擂台记下标、不越界、不破坏原数组）；main 6 分（定义数组并正确传参）。
